function slider(o){
  var cur=o.value, wrap=document.createElement('div'); wrap.className='qz-sl-wrap';
  wrap.innerHTML='<div class="qz-sl-val"></div><div class="qz-sl-track-wrap"><div class="qz-sl-track"><div class="qz-sl-fill"></div><div class="qz-sl-thumb"></div></div></div>'+
    (o.ends?'<div class="qz-sl-ends"><span>'+esc(o.ends[0])+'</span><span>'+esc(o.ends[1])+'</span></div>':'');
  var val=wrap.querySelector('.qz-sl-val'), tw=wrap.querySelector('.qz-sl-track-wrap'), track=wrap.querySelector('.qz-sl-track'),
      fill=wrap.querySelector('.qz-sl-fill'), thumb=wrap.querySelector('.qz-sl-thumb');
  function draw(){ var p=(cur-o.min)/(o.max-o.min)*100; val.textContent=cur+(o.unit?' '+o.unit:''); fill.style.width=p+'%'; thumb.style.left=p+'%'; }
  function fromX(x){ var r=track.getBoundingClientRect(); var v=Math.round(((x-r.left)/r.width*(o.max-o.min)+o.min)/o.step)*o.step; return Math.min(o.max,Math.max(o.min,v)); }
  function set(v){ if(v!==cur){ cur=v; draw(); if(navigator.vibrate) navigator.vibrate(6); } }
  var drag=false;
  tw.addEventListener('pointerdown',function(e){ drag=true; tw.setPointerCapture&&tw.setPointerCapture(e.pointerId); set(fromX(e.clientX)); });
  tw.addEventListener('pointermove',function(e){ if(drag) set(fromX(e.clientX)); });
  tw.addEventListener('pointerup',function(){ drag=false; });
  tw.addEventListener('pointercancel',function(){ drag=false; });
  wrap.getValue=function(){ return cur; }; draw(); return wrap;
}

function gauge(value,sub){
  var ns='http://www.w3.org/2000/svg', wrap=document.createElement('div'); wrap.className='qz-gg-wrap';
  var svg=document.createElementNS(ns,'svg'); svg.setAttribute('viewBox','0 0 200 110'); svg.setAttribute('class','qz-gg-svg');
  svg.innerHTML='<defs><linearGradient id="qzgg" x1="0%" y1="0%" x2="100%" y2="0%"><stop offset="0%" stop-color="#E0574A"/><stop offset="55%" stop-color="#F5A623"/><stop offset="100%" stop-color="#1F9D6B"/></linearGradient></defs>'+
    '<path d="M20,100 A80,80 0 0,1 180,100" fill="none" stroke="#DCE9F4" stroke-width="14" stroke-linecap="round"/>'+
    '<path id="qzArc" fill="none" stroke="url(#qzgg)" stroke-width="14" stroke-linecap="round"/>'+
    '<line id="qzNeedle" x1="100" y1="100" stroke="#0F3157" stroke-width="3" stroke-linecap="round"/><circle cx="100" cy="100" r="6" fill="#0F3157"/>';
  var v=document.createElement('div'); v.className='qz-gg-value'; v.textContent='0';
  var sb=document.createElement('div'); sb.className='qz-gg-sub'; sb.textContent=sub;
  wrap.appendChild(svg); wrap.appendChild(v); wrap.appendChild(sb);
  function pt(a){ var r=(a-180)*Math.PI/180; return [100+80*Math.cos(r),100+80*Math.sin(r)]; }
  function upd(f){ var ang=f*value/100*180, e=pt(ang), arc=svg.querySelector('#qzArc'), nd=svg.querySelector('#qzNeedle');
    arc.setAttribute('d', f<0.01?'':'M20,100 A80,80 0 0,1 '+e[0]+','+e[1]); var n=pt(ang); nd.setAttribute('x2',100+(n[0]-100)*.9); nd.setAttribute('y2',100+(n[1]-100)*.9);
    v.textContent=Math.round(f*value); }
  upd(0); var st=null; function fr(t){ if(!st) st=t; var k=Math.min((t-st)/1400,1); upd(1-Math.pow(1-k,3)); if(k<1) requestAnimationFrame(fr); }
  setTimeout(function(){ requestAnimationFrame(fr); },250);
  setTimeout(function(){ upd(1); },1900); 
  return wrap;
}
