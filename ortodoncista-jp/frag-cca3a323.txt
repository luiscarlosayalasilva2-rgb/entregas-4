  var seg=parseInt(A.seguridad,10)||5, casos=parseInt(A.casos,10)||8;
  var dudas=Array.isArray(A.dudas)?A.dudas:[];
  var bonus={'2年未満':0,'2〜5年':4,'5〜10年':7,'10年以上':9}[A.anios]||4;
  var idx=Math.max(18,Math.min(84, Math.round(seg*7.5+bonus-Math.min(dudas.length,6)*2.5)));
  var perfil = seg<=4 ? 'プレッシャーの中で判断するタイプ' : (seg<=7 ? '判断の型をつくっている途中のタイプ' : 'もっと速く判断したいタイプ');
  var lista = dudas.length ? dudas : ['抜歯か、非抜歯か','II級','仕上げと正中線'];
  var items = lista.map(function(d){ var r=CONFIG.RESUELVE[d]; if(!r) return '';
    return '<div class="qz-mi"><img loading="lazy" src="'+CONFIG.MAPAS[r.map]+'" alt="ビジュアルマップ：'+esc(d)+'"><div><b>'+esc(d)+'</b><span>'+esc(r.r)+' · ビジュアルマップ付き</span></div></div>'; }).join('');
  var notas='';
  if(A.alargue && A.alargue!=='いいえ、一度もない') notas+='<div class="qz-note">⏱ <b>長引く治療：</b>治療計画ミス・レーダーが症例を長引かせるつまずきを示すので、ブラケットを装着する前に気づけます。</div>';
  if(A.reaccion==='始めてから、途中で修正する'||A.reaccion==='確信がなくても、その場で判断する') notas+='<div class="qz-note">🧭 <b>その場で判断している：</b>4ステップ臨床プランナーがあれば、診断・治療目標・メカニクスを整理したうえで患者さんを迎えられます。</div>';
  notas+='<div class="qz-note">📅 難しい症例を<b>月に約'+casos+'件</b>診ているなら、<b>年間で約'+(casos*12)+'回の判断</b>が、いまはその場の記憶に頼っていることになります。</div>';
  stage.innerHTML='<div class="qz-result">'+
    '<span class="qz-phase">先生の臨床マップ'+(name?' · '+name+' 先生':'')+'</span>'+
    '<h1 class="qz-h">'+(name?name+' 先生のタイプ：':'先生のタイプ：')+'<br><span class="am">'+perfil+'</span></h1>'+
    '<div id="qzGauge"></div>'+
    '<div class="qz-vs"><div class="now"><span class="lab">現在の自信度</span><div class="v">'+idx+'/100</div></div>'+
    '<div class="goal"><span class="lab">システム導入後</span><div class="v">90+/100</div></div></div>'+
    '<p class="qz-sub" style="margin-bottom:6px">足りないのは知識ではなく、整理された判断がすぐ手元にあることです。いま迷っている症例ごとに、次の内容を確認できます。</p>'+
    '<div class="qz-map"><div class="qz-map-h">迷いごとの解決先</div>'+items+'</div>'+
    notas+
    '<button class="qz-cta" id="qzSeeOffer">バイブルへのアクセスを見る</button></div>';
  document.getElementById('qzGauge').appendChild(gauge(idx,'現在の臨床判断の自信度（100点満点）'));
  stage.querySelectorAll('.qz-mi').forEach(function(z,k){ setTimeout(function(){ z.classList.add('in'); },600+k*140); });
  document.getElementById('qzSeeOffer').onclick=renderSale;
  window.scrollTo(0,0);
}
